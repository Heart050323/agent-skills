\newcommand{\bookcover}[4]{% #1 科目名 #2 副題 #3 説明 #4 日付
\begin{titlepage}
\begin{tikzpicture}[remember picture,overlay]
  \fill[blue!45!black] (current page.north west) rectangle ([yshift=-95mm]current page.north east);
  \fill[blue!25!black] ([yshift=-95mm]current page.north west) rectangle ([yshift=-99mm]current page.north east);
  \node[anchor=south west,text=white,font=\sffamily\bfseries\fontsize{38}{44}\selectfont]
    at ([xshift=22mm,yshift=-80mm]current page.north west) {#1};
  \node[anchor=north west,text width=165mm,font=\Large\bfseries,text=blue!45!black]
    at ([xshift=22mm,yshift=-110mm]current page.north west) {#2};
  \node[anchor=north west,text width=165mm,font=\large,text=black!75]
    at ([xshift=22mm,yshift=-128mm]current page.north west) {#3};
  \draw[blue!45!black,line width=1pt] ([xshift=22mm,yshift=40mm]current page.south west) -- ([xshift=-22mm,yshift=40mm]current page.south east);
  \node[anchor=north west,font=\normalsize,text=black!70]
    at ([xshift=22mm,yshift=36mm]current page.south west) {#4};
\end{tikzpicture}
\end{titlepage}}
